cmake_minimum_required(VERSION 3.20)
project(ffamp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/*.cpp src/opt/*.cpp)
add_library(ffamp_lib STATIC ${CORE_SOURCES})
set_target_properties(ffamp_lib PROPERTIES OUTPUT_NAME ffamp)
if(MSVC)
    set_target_properties(ffamp_lib PROPERTIES OUTPUT_NAME libffamp)
endif()
target_include_directories(ffamp_lib PUBLIC include PRIVATE src)
if(MSVC)
    target_compile_options(ffamp_lib PRIVATE /W3 /EHsc /clang:-fwrapv)
    target_compile_definitions(ffamp_lib PUBLIC _USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    get_filename_component(LLVM_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    file(GLOB CLANG_BUILTINS "${LLVM_BIN}/../lib/clang/*/lib/windows/clang_rt.builtins-x86_64.lib")
    if(CLANG_BUILTINS)
        list(GET CLANG_BUILTINS 0 CLANG_BUILTINS)
        target_link_libraries(ffamp_lib PUBLIC "${CLANG_BUILTINS}")
    endif()
else()
    target_compile_options(ffamp_lib PRIVATE -Wall -Wextra -fwrapv)
endif()

function(absent_optional name define header)
    find_library(${name}_LIBRARY NAMES ${ARGN})
    find_path(${name}_INCLUDE ${header})
    if(${name}_LIBRARY AND ${name}_INCLUDE)
        target_include_directories(ffamp_lib PRIVATE ${${name}_INCLUDE})
        target_link_libraries(ffamp_lib PUBLIC ${${name}_LIBRARY})
        target_compile_definitions(ffamp_lib PUBLIC ${define})
        message(STATUS "FFamp: ${name} found")
    else()
        message(STATUS "FFamp: ${name} not found, skipped")
    endif()
endfunction()

option(ABSENT_FUZZ "Build the library and the fuzz targets with coverage and undefined-behaviour checks" OFF)
option(ABSENT_LIBFUZZER "Use the compiler's own fuzzing engine instead of tests/fuzz/driver.cpp" OFF)
if(NOT ANDROID AND NOT MSVC)
    if(APPLE)
        set(FUZZ_CHECKS -g -O1 -fno-omit-frame-pointer -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all -D_LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_DEBUG)
    else()
        set(FUZZ_CHECKS -g -O1 -fno-omit-frame-pointer -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
    endif()
    if(ABSENT_FUZZ AND ABSENT_LIBFUZZER)
        list(APPEND FUZZ_CHECKS -fsanitize=fuzzer-no-link)
    elseif(ABSENT_FUZZ)
        list(APPEND FUZZ_CHECKS -fsanitize-coverage=trace-pc-guard)
    endif()
    if(ABSENT_FUZZ)
        target_compile_options(ffamp_lib PRIVATE ${FUZZ_CHECKS})
    endif()
    foreach(target open packets probe audio video resample mpeg4 wma vc1 msmpeg4)
        add_executable(fuzz_${target} tests/fuzz/targets.cpp)
        target_include_directories(fuzz_${target} PRIVATE src)
        target_compile_definitions(fuzz_${target} PRIVATE FUZZ_TARGET=${target})
        target_compile_options(fuzz_${target} PRIVATE -Wall -Wextra -Wno-unused-function)
        target_link_libraries(fuzz_${target} PRIVATE ffamp_lib)
        if(ABSENT_FUZZ)
            target_compile_options(fuzz_${target} PRIVATE ${FUZZ_CHECKS})
            target_link_options(fuzz_${target} PRIVATE -fsanitize=undefined,float-cast-overflow)
        endif()
        if(ABSENT_FUZZ AND ABSENT_LIBFUZZER)
            target_link_options(fuzz_${target} PRIVATE -fsanitize=fuzzer)
        else()
            target_sources(fuzz_${target} PRIVATE tests/fuzz/driver.cpp)
            if(ABSENT_FUZZ)
                set_source_files_properties(tests/fuzz/driver.cpp PROPERTIES COMPILE_OPTIONS "-fno-sanitize-coverage=trace-pc-guard;-fno-sanitize=all")
            endif()
        endif()
    endforeach()
endif()

    foreach(name ffmpeg ffprobe)

        set_target_properties(shim_${name} PROPERTIES OUTPUT_NAME ${name} RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/shim)
    endforeach()
endif()
