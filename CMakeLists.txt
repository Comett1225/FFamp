cmake_minimum_required(VERSION 3.20)
project(absent_media LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_library(absentmedia STATIC src/mp4_read.cpp src/mp4_write.cpp src/mkv_write.cpp)
target_include_directories(absentmedia PUBLIC include PRIVATE src)
target_compile_options(absentmedia PRIVATE -Wall -Wextra)

find_library(OPUS_LIBRARY NAMES libopus.a opus)
find_path(OPUS_INCLUDE opus/opus.h)
if(OPUS_LIBRARY AND OPUS_INCLUDE)
    target_sources(absentmedia PRIVATE src/opus_encode.cpp)
    target_include_directories(absentmedia PRIVATE ${OPUS_INCLUDE})
    target_link_libraries(absentmedia PUBLIC ${OPUS_LIBRARY})
    target_compile_definitions(absentmedia PUBLIC ABSENT_OPUS)
endif()

if(APPLE)
    target_sources(absentmedia PRIVATE src/mac/vt_decode.cpp src/mac/vt_encode.cpp src/mac/at_aac.cpp)
    target_link_libraries(absentmedia PUBLIC "-framework VideoToolbox" "-framework CoreMedia" "-framework CoreVideo" "-framework CoreFoundation" "-framework AudioToolbox")
    add_executable(amedia tools/amedia.cpp)
    target_link_libraries(amedia PRIVATE absentmedia)
endif()
