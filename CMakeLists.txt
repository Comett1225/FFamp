cmake_minimum_required(VERSION 3.20)
project(absent_media LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/*.cpp src/opt/*.cpp)
add_library(absentmedia STATIC ${CORE_SOURCES})
target_include_directories(absentmedia PUBLIC include PRIVATE src)
target_compile_options(absentmedia PRIVATE -Wall -Wextra)

function(absent_optional name define header)
    find_library(${name}_LIBRARY NAMES ${ARGN})
    find_path(${name}_INCLUDE ${header})
    if(${name}_LIBRARY AND ${name}_INCLUDE)
        target_include_directories(absentmedia PRIVATE ${${name}_INCLUDE})
        target_link_libraries(absentmedia PUBLIC ${${name}_LIBRARY})
        target_compile_definitions(absentmedia PUBLIC ${define})
        message(STATUS "absent-media: ${name} found")
    else()
        message(STATUS "absent-media: ${name} not found, skipped")
    endif()
endfunction()

absent_optional(OPUS ABSENT_OPUS opus/opus.h libopus.a opus)
absent_optional(LAME ABSENT_LAME lame/lame.h libmp3lame.a mp3lame)
absent_optional(DAV1D ABSENT_DAV1D dav1d/dav1d.h dav1d)
absent_optional(VPX ABSENT_VPX vpx/vpx_decoder.h libvpx.a vpx)
absent_optional(SVTAV1 ABSENT_SVTAV1 svt-av1/EbSvtAv1Enc.h SvtAv1Enc)

if(APPLE)
    file(GLOB MAC_SOURCES CONFIGURE_DEPENDS src/mac/*.cpp src/mac/*.mm)
    target_sources(absentmedia PRIVATE ${MAC_SOURCES})
    target_link_libraries(absentmedia PUBLIC "-framework VideoToolbox" "-framework CoreMedia" "-framework CoreVideo" "-framework CoreFoundation" "-framework AudioToolbox" "-framework Foundation" "-framework AVFoundation" "-framework ImageIO" "-framework CoreGraphics")
    file(GLOB TOOL_SOURCES CONFIGURE_DEPENDS tools/*.cpp)
    add_executable(amedia ${TOOL_SOURCES})
    target_link_libraries(amedia PRIVATE absentmedia)
endif()
